2️⃣  RUN THIS ONE COMMAND (copy all 8 lines together):
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

cd studo-ecommerce && \
npm install --production && \
bash deploy-vps.sh && \
pm2 start npm --name "studo-ecommerce" -- start -- -p 3002 && \
pm2 save && \
echo "✅ DONE! Visit http://31.97.72.28:3002"


3️⃣  SETUP NGINX (copy all 12 lines together):
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
cat > /etc/nginx/sites-available/studo-ecommerce << 'NGINXEOF'
server {
    listen 80;
    server_name studo.ps www.studo.ps 31.97.72.28;
    location / {
        proxy_pass http://localhost:3002;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }
}
NGINXEOF
ln -sf /etc/nginx/sites-available/studo-ecommerce /etc/nginx/sites-enabled/ && \
nginx -t && \
systemctl reload nginx && \
echo "✅ Nginx configured! Visit http://studo.ps"
